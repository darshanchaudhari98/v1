clc; 
clear; 
close all; 
SOC0 = 100; 
Q = 100; 
I_values = [20 50 80]; 
t = 0:0.01:10; 
figure; 
for k = 1:length(I_values) 
    I = I_values(k); 
    SOC = SOC0 - (I*t/Q)*100; 
    SOC(SOC < 0) = 0; 
    plot(t,SOC,'LineWidth',2); 
    hold on; 
end 
grid on; 
xlabel('Time (hour)'); 
ylabel('SOC (%)'); 
title('SOC Estimation for Different Discharge Currents'); 
legend('20 A','50 A','80 A');
